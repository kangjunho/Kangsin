D) GO와 KEGG enrichment

GO에서는 Biological Process, KEGG에서는 mouse organism code mmu를 사용합니다. 이번 실습에서는 유의하지 않은 항목까지 반환받아 결과 구조를 확인할 수 있도록 pvalueCutoff와 qvalueCutoff를 1로 지정합니다.

run_go <- function(gene_ids) {
  if (length(gene_ids) < 10) return(NULL)

  clusterProfiler::enrichGO(
    gene = gene_ids,
    universe = universe_entrez,
    OrgDb = org.Mm.eg.db,
    keyType = "ENTREZID",
    ont = "BP",
    pAdjustMethod = "BH",
    pvalueCutoff = 1,
    qvalueCutoff = 1,
    readable = TRUE
  )
}

run_kegg <- function(gene_ids) {
  if (length(gene_ids) < 10) return(NULL)

  clusterProfiler::enrichKEGG(
    gene = gene_ids,
    universe = universe_entrez,
    organism = "mmu",
    keyType = "kegg",
    pAdjustMethod = "BH",
    pvalueCutoff = 1,
    qvalueCutoff = 1
  )
}

go_up <- run_go(up_entrez)
go_down <- run_go(down_entrez)
kegg_up <- run_kegg(up_entrez)
kegg_down <- run_kegg(down_entrez)

F) Cell–cell interaction 분석 개념

CellChat은 알려진 ligand–receptor database와 세포형별 발현을 이용해 sender–receiver 관계를 추론합니다. 본 데이터는 마우스이므로 CellChatDB.mouse를 사용하고, 실습에서는 계산 시간을 줄이기 위해 조건별·세포형별 최대 300개 세포를 사용합니다.

용어의미
Sender특정 ligand를 발현하여 신호를 보내는 세포형
Receiver해당 receptor를 발현하여 신호를 받는 세포형
Interaction number통계 기준을 통과한 ligand–receptor 연결의 수
Interaction strengthCellChat이 추정한 communication probability의 종합 크기

safe_enrichment_plot <- function(
    enrichment_result,
    plot_title,
    dot_color
) {

  result_table <- if (
    is.null(enrichment_result)
  ) {
    data.frame()
  } else {
    as.data.frame(enrichment_result)
  }

  if (nrow(result_table) == 0) {
    return(
      ggplot() +
        annotate(
          "text",
          x = 0,
          y = 0,
          label = "No enriched term"
        ) +
        xlim(-1, 1) +
        ylim(-1, 1) +
        labs(title = plot_title) +
        theme_void() +
        theme(
          plot.title = element_text(
            face = "bold",
            hjust = 0.5
          )
        )
    )
  }

  enrichplot::dotplot(
    enrichment_result,
    showCategory = 8,
    color = "p.adjust"
  ) +
    ggtitle(plot_title) +
    scale_color_gradient(
      low = dot_color,
      high = "grey85",
      trans = "reverse"
    ) +
    theme(
      plot.title = element_text(
        face = "bold",
        hjust = 0.5
      ),
      axis.text.y = element_text(size = 9)
    )
}

# 계산 시간을 줄이기 위해 condition별·cell type별 최대 300개 세포 사용

run_cellchat <- function(
    seurat_object,
    condition_name,
    max_cells_per_type = 300
) {

  condition_cells <- colnames(seurat_object)[
    seurat_object$condition == condition_name &
      seurat_object$celltype_intermediate != "Unresolved"
  ]

  condition_object <- subset(
    seurat_object,
    cells = condition_cells
  )

  cells_by_type <- split(
    colnames(condition_object),
    condition_object$celltype_intermediate
  )

  selected_cells <- unlist(
    lapply(
      cells_by_type,
      function(current_cells) {
        sample(
          current_cells,
          size = min(
            length(current_cells),
            max_cells_per_type
          )
        )
      }
    ),
    use.names = FALSE
  )

  expression_matrix <- Seurat::GetAssayData(
    condition_object,
    assay = "RNA",
    slot = "data"
  )[
    ,
    selected_cells,
    drop = FALSE
  ]

  cellchat_metadata <- data.frame(
    celltype = condition_object$celltype_intermediate[
      match(
        selected_cells,
        colnames(condition_object)
      )
    ],
    row.names = selected_cells
  )

  cellchat_object <- CellChat::createCellChat(
    object = expression_matrix,
    meta = cellchat_metadata,
    group.by = "celltype"
  )

  cellchat_database <- CellChat::subsetDB(
    CellChat::CellChatDB.mouse,
    search = "Secreted Signaling"
  )

  cellchat_object@DB <- cellchat_database

  cellchat_object <- CellChat::subsetData(
    cellchat_object
  )

  cellchat_object <- CellChat::identifyOverExpressedGenes(
    cellchat_object
  )

  cellchat_object <- CellChat::identifyOverExpressedInteractions(
    cellchat_object
  )

  cellchat_object <- CellChat::computeCommunProb(
    cellchat_object,
    type = "triMean",
    population.size = TRUE
  )

  cellchat_object <- CellChat::filterCommunication(
    cellchat_object,
    min.cells = 10
  )

  cellchat_object <- CellChat::computeCommunProbPathway(
    cellchat_object
  )

  cellchat_object <- CellChat::aggregateNet(
    cellchat_object
  )

  cellchat_object
}

  1. GO·KEGG 2×2 panel에서 WD 상향과 하향 결과의 주요 biological theme을 각각 설명합니다.
  2. Macrophage가 아닌 세포형의 DEG를 사용해 functional annotation panel을 다시 만듭니다.
  3. ND와 WD에서 interaction number와 interaction strength가 어떻게 달라지는지 비교합니다.
  4. WD에서 증가한 signaling pathway 하나를 선택하고 주요 sender와 receiver 세포형을 확인합니다.
  5. 선택한 ligand–receptor 관계를 추가로 검증하려면 어떤 실험 또는 데이터가 필요한지 설명합니다.
