sample_information <- integrated_obj@meta.data |>
  dplyr::select(orig.ident, condition) |>
  dplyr::distinct()

celltype_levels <- sort(
  unique(integrated_obj$celltype_intermediate)
)

cell_composition <- integrated_obj@meta.data |>
  dplyr::count(
    orig.ident,
    condition,
    celltype_intermediate,
    name = "cell_count"
  ) |>
  tidyr::complete(
    tidyr::nesting(orig.ident, condition),
    celltype_intermediate = celltype_levels,
    fill = list(cell_count = 0)
  ) |>
  dplyr::group_by(orig.ident, condition) |>
  dplyr::mutate(
    total_cells = sum(cell_count),
    proportion = cell_count / total_cells,
    percentage = proportion * 100
  ) |>
  dplyr::ungroup()

composition_test <- cell_composition |>
  dplyr::group_by(celltype_intermediate) |>
  dplyr::summarise(
    ND_mean_percent = mean(percentage[condition == "ND"]),
    WD_mean_percent = mean(percentage[condition == "WD"]),
    WD_minus_ND = WD_mean_percent - ND_mean_percent,
    p_value = tryCatch(
      wilcox.test(
        percentage[condition == "WD"],
        percentage[condition == "ND"],
        exact = FALSE
      )$p.value,
      error = function(e) NA_real_
    ),
    .groups = "drop"
  ) |>
  dplyr::mutate(p_adj = p.adjust(p_value, method = "BH")) |>
  dplyr::arrange(p_value)

G-2) 모든 중분류 세포형의 DEG 계산

celltype_list <- sort(
  unique(integrated_obj$celltype_intermediate)
)

deg_all_celltypes <- lapply(
  celltype_list,
  function(current_celltype) {
    current_obj <- subset(
      integrated_obj,
      subset = celltype_intermediate == current_celltype
    )

    condition_count <- table(current_obj$condition)
    if (
      !all(c("ND", "WD") %in% names(condition_count)) ||
      any(condition_count[c("ND", "WD")] < 50)
    ) return(NULL)

    DefaultAssay(current_obj) <- "RNA"
    Idents(current_obj) <- "condition"

    current_deg <- FindMarkers(
      current_obj,
      ident.1 = "WD", ident.2 = "ND",
      assay = "RNA", slot = "data",
      test.use = "wilcox", only.pos = FALSE,
      min.pct = 0.10, logfc.threshold = 0.25,
      max.cells.per.ident = 1000,
      random.seed = 1234
    )
    current_deg$gene <- rownames(current_deg)

    current_deg |>
      dplyr::relocate(gene) |>
      dplyr::mutate(
        celltype = current_celltype,
        comparison = "WD_vs_ND",
        direction = dplyr::case_when(
          p_val_adj < 0.05 & avg_log2FC >= 0.25 ~ "Up_in_WD",
          p_val_adj < 0.05 & avg_log2FC <= -0.25 ~ "Down_in_WD",
          TRUE ~ "Not_significant"
        )
      )
  }
)

H) Cell-level DEG와 pseudobulk DEG의 차이

항목Cell-level WilcoxonSample-level pseudobulk
분석 단위개별 세포각 sample에서 합산한 raw counts
장점빠르고 직관적이며 탐색에 유용생물학적 반복을 올바르게 반영
주요 한계동일 mouse의 세포를 독립 반복처럼 취급할 위험sample 수가 적으면 검정력이 제한됨
권장 용도교육·탐색·marker 후보 확인조건 간 DEG의 통계적 추론과 논문 결과

Pseudobulk는 동일한 cell type 안에서 각 mouse의 raw count를 유전자별로 합산하여 mouse 하나를 하나의 RNA-seq sample처럼 분석합니다. 따라서 ND 3개와 WD 3개가 실제 반복수입니다. 반면 cell-level 결과는 세포 상태의 방향을 빠르게 찾는 데 유용하지만 p-value가 지나치게 작아질 수 있습니다.

  • AggregateExpression()으로 sample별 raw counts를 합산합니다.
  • edgeR에서는 library size 보정, dispersion 추정과 sample-level 통계 검정을 수행합니다.
  • 두 방법에서 방향이 일치하고 여러 sample에서 일관된 유전자를 우선 해석합니다.
  • 특정 sample 하나가 결과를 주도하거나 sample당 세포 수가 너무 적으면 결론을 보류합니다.

참고: Seurat differential expression 및 pseudobulk vignette, biological replicate를 고려한 single-cell DEG 연구.

deg_all_celltypes <- lapply(
  celltype_list,
  function(current_celltype) {

    current_obj <- subset(
      integrated_obj,
      subset =
        celltype_intermediate == current_celltype
    )

    sample_cell_count <- table(
      current_obj$orig.ident,
      current_obj$condition
    )

    condition_count <- table(
      current_obj$condition
    )

    # 두 조건 모두 최소 50개 세포가 있을 때만 실행
    if (
      !all(c("ND", "WD") %in% names(condition_count)) ||
      any(condition_count[c("ND", "WD")] < 50)
    ) {
      return(NULL)
    }

    DefaultAssay(current_obj) <- "RNA"
    Idents(current_obj) <- "condition"

    current_deg <- FindMarkers(
      current_obj,
      ident.1 = "WD",
      ident.2 = "ND",
      assay = "RNA",
      slot = "data",
      test.use = "wilcox",
      only.pos = FALSE,
      min.pct = 0.10,
      logfc.threshold = 0.25,
      max.cells.per.ident = 1000,
      random.seed = 1234
    )

    current_deg$gene <- rownames(current_deg)

    current_deg |>
      dplyr::relocate(gene) |>
      dplyr::mutate(
        celltype = current_celltype,
        comparison = "WD_vs_ND",
        direction = dplyr::case_when(
          p_val_adj < 0.05 &
            avg_log2FC >= 0.25 ~ "Up_in_WD",
          p_val_adj < 0.05 &
            avg_log2FC <= -0.25 ~ "Down_in_WD",
          TRUE ~ "Not_significant"
        )
      )
  }
)

# 7. 권장 분석: sample-level pseudobulk DEG with edgeR
# ============================================================
# Cell-level Wilcoxon은 실습과 탐색에는 편리하지만,
# 같은 mouse에서 나온 세포를 독립 반복으로 취급할 수 있습니다.
# 논문용 추론에서는 sample을 반복 단위로 하는 pseudobulk를 권장합니다.

target_celltype_pb <- "Macrophage"

  pb_obj <- subset(
    integrated_obj,
    subset =
      celltype_intermediate == target_celltype_pb
  )

  pb_cell_count <- table(
    pb_obj$orig.ident
  )

  valid_samples <- names(
    pb_cell_count[pb_cell_count >= 20]
  )

  pb_obj <- subset(
    pb_obj,
    subset = orig.ident %in% valid_samples
  )

  pb_counts <- AggregateExpression(
    pb_obj,
    assays = "RNA",
    group.by = "orig.ident",
    slot = "counts",
    return.seurat = FALSE
  )$RNA

pb_metadata <- pb_obj@meta.data |>
    dplyr::select(
      orig.ident,
      condition
    ) |>
    dplyr::distinct()

  pb_metadata <- pb_metadata[
    match(
      colnames(pb_counts),
      pb_metadata$orig.ident
    ),
    ,
    drop = FALSE
  ]

  stopifnot(
    all(
      colnames(pb_counts) ==
        pb_metadata$orig.ident
    )
  )

  group <- factor(
    pb_metadata$condition,
    levels = c("ND", "WD")
  )

  y <- edgeR::DGEList(
    counts = pb_counts,
    group = group
  )

  keep <- edgeR::filterByExpr(
    y,
    group = group
  )

  y <- y[keep, , keep.lib.sizes = FALSE]
  y <- edgeR::calcNormFactors(y)

  design <- model.matrix(
    ~ group
  )

  y <- edgeR::estimateDisp(
    y,
    design
  )

  fit <- edgeR::glmQLFit(
    y,
    design,
    robust = TRUE
  )

  qlf <- edgeR::glmQLFTest(
    fit,
    coef = "groupWD"
  )

  pseudobulk_deg <- edgeR::topTags(
    qlf,
    n = Inf
  )$table

  pseudobulk_deg$gene <- rownames(
    pseudobulk_deg
  )

  pseudobulk_deg <- pseudobulk_deg |>
    dplyr::relocate(gene)

print(
  head(pseudobulk_deg, 20)
)
