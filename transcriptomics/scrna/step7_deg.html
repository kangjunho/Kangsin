Part 1. 이론적 배경

A) 세포 비율과 DEG는 다른 질문이다

분석비교 대상해석
세포 조성각 sample에서 세포형이 차지하는 비율세포 모집, 소실 또는 조직 구성 변화

H) 왜 pseudobulk가 필요한가?

Cell-level Wilcoxon은 빠르고 교육용 탐색에 편리하지만 같은 mouse에서 나온 수천 개 세포를 독립 반복처럼 취급할 수 있습니다. 논문용 조건 간 DEG에서는 각 cell type의 counts를 sample별로 합산하고 sample을 반복 단위로 분석하는 pseudobulk가 권장됩니다.

  • AggregateExpression(): 동일 sample·cell type의 raw counts를 합산
  • edgeR: ND 3개와 WD 3개 pseudobulk profile을 비교
  • 세포 수가 지나치게 적은 sample-cell type 조합은 제외하거나 결과를 보류

참고: Seurat differential expression 및 pseudobulk vignette, biological replicate를 고려한 single-cell DEG 연구.

deg_all_celltypes <- lapply(
  celltype_list,
  function(current_celltype) {

    current_obj <- subset(
      integrated_obj,
      subset =
        celltype_intermediate == current_celltype
    )

    sample_cell_count <- table(
      current_obj$orig.ident,
      current_obj$condition
    )

    condition_count <- table(
      current_obj$condition
    )

    # 두 조건 모두 최소 50개 세포가 있을 때만 실행
    if (
      !all(c("ND", "WD") %in% names(condition_count)) ||
      any(condition_count[c("ND", "WD")] < 50)
    ) {
      return(NULL)
    }

    DefaultAssay(current_obj) <- "RNA"
    Idents(current_obj) <- "condition"

    current_deg <- FindMarkers(
      current_obj,
      ident.1 = "WD",
      ident.2 = "ND",
      assay = "RNA",
      slot = "data",
      test.use = "wilcox",
      only.pos = FALSE,
      min.pct = 0.10,
      logfc.threshold = 0.25,
      max.cells.per.ident = 1000,
      random.seed = 1234
    )

    current_deg$gene <- rownames(current_deg)

    current_deg |>
      relocate(gene) |>
      mutate(
        celltype = current_celltype,
        comparison = "WD_vs_ND",
        direction = case_when(
          p_val_adj < 0.05 &
            avg_log2FC >= 0.25 ~ "Up_in_WD",
          p_val_adj < 0.05 &
            avg_log2FC <= -0.25 ~ "Down_in_WD",
          TRUE ~ "Not_significant"

  }

# ============================================================
# 7. 권장 분석: sample-level pseudobulk DEG with edgeR
# ============================================================
# Cell-level Wilcoxon은 실습과 탐색에는 편리하지만,
# 같은 mouse에서 나온 세포를 독립 반복으로 취급할 수 있습니다.
# 논문용 추론에서는 sample을 반복 단위로 하는 pseudobulk를 권장합니다.

target_celltype_pb <- "Macrophage"

  pb_obj <- subset(
    integrated_obj,
    subset =
      celltype_intermediate == target_celltype_pb
  )

  pb_cell_count <- table(
    pb_obj$orig.ident
  )

  valid_samples <- names(
    pb_cell_count[pb_cell_count >= 20]
  )

  pb_obj <- subset(
    pb_obj,
    subset = orig.ident %in% valid_samples
  )

  pb_counts <- AggregateExpression(
    pb_obj,
    assays = "RNA",
    group.by = "orig.ident",
    slot = "counts",
    return.seurat = FALSE
  )$RNA

pb_metadata <- pb_obj@meta.data |>
    dplyr::select(
      orig.ident,
      condition
    ) |>
    dplyr::distinct()

  pb_metadata <- pb_metadata[
    match(
      colnames(pb_counts),
      pb_metadata$orig.ident
    ),
    ,
    drop = FALSE
  ]

  stopifnot(
    all(
      colnames(pb_counts) ==
        pb_metadata$orig.ident
    )
  )

  group <- factor(
    pb_metadata$condition,
    levels = c("ND", "WD")
  )

  y <- edgeR::DGEList(
    counts = pb_counts,
    group = group
  )

  keep <- edgeR::filterByExpr(
    y,
    group = group
  )

  y <- y[keep, , keep.lib.sizes = FALSE]
  y <- edgeR::calcNormFactors(y)

  design <- model.matrix(
    ~ group
  )

  y <- edgeR::estimateDisp(
    y,
    design
  )

  fit <- edgeR::glmQLFit(
    y,
    design,
    robust = TRUE
  )

  qlf <- edgeR::glmQLFTest(
    fit,
    coef = "groupWD"
  )

  pseudobulk_deg <- edgeR::topTags(
    qlf,
    n = Inf
  )$table

  pseudobulk_deg$gene <- rownames(
    pseudobulk_deg
  )

  pseudobulk_deg <- pseudobulk_deg |>
    relocate(gene)

print(
  head(pseudobulk_deg, 20)
)
